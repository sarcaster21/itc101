import random

activities = [
    "Read a book",
    "Go for a walk",
    "Listen to music",
    "Watch a movie",
    "Draw something",
    "Clean your room",
    "Learn something new",
    "Call a friend",
    "Do some exercise",
    "Cook something new"
]

types = [
    "education",
    "recreational",
    "social",
    "relaxation",
    "creative"
]

activity = random.choice(activities)
activity_type = random.choice(types)
participants = random.randint(1, 4)

print()
print("Your activity:")
print(activity)
print("Type:", activity_type)
print("Participants:", participants)
