# Simple Bank Account Management System

# Function to display menu
def display_menu():
    print("\n========== BANK MENU ==========")
    print("1. Check Balance")
    print("2. Deposit Money")
    print("3. Withdraw Money")
    print("4. Account Information")
    print("5. Exit")


# Taking account holder name
name = input("Enter account holder name: ")

# Taking and validating 12-digit account number
while True:
    account_number = input("Enter 12-digit account number: ")

    if account_number.isdigit() and len(account_number) == 12:
        break
    else:
        print("Invalid account number!")
        print("Account number must contain exactly 12 digits.")


# Initial balance
balance = 5000.0

# Tuple containing account information
account_info = (name, account_number)

# Set containing available account types
account_types = {"Savings", "Current"}

print("\nAvailable Account Types:", account_types)

account_type = input("Enter account type: ")

# Convert first letter to uppercase
account_type = account_type.capitalize()

# Check account type
if account_type not in account_types:
    print("Invalid account type! Defaulting to Savings.")
    account_type = "Savings"


# Infinite loop for menu
while True:

    display_menu()

    try:
        choice = int(input("Enter your choice: "))

        if choice == 1:

            # Checking balance
            print("\nCurrent Balance: ₹", balance)

        elif choice == 2:

            # Deposit
            amount = float(input("Enter amount to deposit: ₹"))

            if amount > 0:
                balance += amount
                print("Amount deposited successfully.")
                print("New Balance: ₹", balance)
            else:
                print("Amount must be greater than zero.")

        elif choice == 3:

            # Withdrawal
            amount = float(input("Enter amount to withdraw: ₹"))

            if amount <= 0:
                print("Amount must be greater than zero.")

            elif amount > balance:
                print("Insufficient balance.")

            else:
                balance -= amount
                print("Withdrawal successful.")
                print("Remaining Balance: ₹", balance)

        elif choice == 4:

            # Display account information
            print("\n========== ACCOUNT INFORMATION ==========")
            print("Name           :", account_info[0])
            print("Account Number :", account_info[1])
            print("Account Type   :", account_type)
            print("Balance        : ₹", balance)

        elif choice == 5:

            print("\nThank you for using our banking system!")
            break

        else:
            print("Invalid choice. Please select 1-5.")

    except ValueError:
        print("Please enter a valid number.")
